cmake_minimum_required(VERSION 3.28)
project(BranchTruceEditorExperiment LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.6 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2)
find_package(KF6Kirigami 6.0 REQUIRED)
add_executable(editor-experiment main.cpp)
target_link_libraries(editor-experiment PRIVATE Qt6::Quick Qt6::QuickControls2)
target_compile_options(editor-experiment PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Werror)
qt_add_resources(editor-experiment scene PREFIX / FILES Editor.qml)
message(STATUS "Experiment: Qt ${Qt6_VERSION}, Kirigami ${KF6Kirigami_VERSION}")
