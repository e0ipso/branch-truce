cmake_minimum_required(VERSION 3.28)
project(BranchTruceNativeReplay LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 6.6 REQUIRED COMPONENTS Core Gui Quick Network Test)
add_library(native_replay MODULE Driver.cpp)
target_link_libraries(native_replay PRIVATE Qt6::Core Qt6::Gui Qt6::Quick Qt6::Network Qt6::Test)
target_compile_options(native_replay PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(native-replay-image ImageCheck.cpp)
target_link_libraries(native-replay-image PRIVATE Qt6::Core Qt6::Gui)
target_compile_options(native-replay-image PRIVATE -Wall -Wextra -Wpedantic -Werror)
